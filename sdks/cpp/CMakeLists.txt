cmake_minimum_required(VERSION 3.16)
project(glance_widgets_sdk VERSION 1.0.0 DESCRIPTION "SDK for generating Glance Widgets API responses" LANGUAGES CXX)
include(GNUInstallDirs)
add_library(glance_widgets INTERFACE)
add_library(GlanceWidgets::SDK ALIAS glance_widgets)
set_target_properties(glance_widgets PROPERTIES EXPORT_NAME SDK)
target_include_directories(glance_widgets INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(glance_widgets INTERFACE cxx_std_17)
# Floating-point results must match the TypeScript reference: never fuse a*b+c.
target_compile_options(glance_widgets INTERFACE $<$<OR:$<CXX_COMPILER_ID:GNU>,$<CXX_COMPILER_ID:Clang>,$<CXX_COMPILER_ID:AppleClang>>:-ffp-contract=off>)
include(CTest)
if(BUILD_TESTING)
  add_executable(glance_widgets_geometry tests/test_geometry.cpp)
  target_link_libraries(glance_widgets_geometry PRIVATE glance_widgets)
  add_test(NAME geometry_api COMMAND glance_widgets_geometry)
  add_executable(glance_widgets_signature tests/test_signature.cpp)
  target_link_libraries(glance_widgets_signature PRIVATE glance_widgets)
  add_test(NAME signature_api COMMAND glance_widgets_signature)
  add_executable(glance_widgets_color tests/test_color.cpp)
  target_link_libraries(glance_widgets_color PRIVATE glance_widgets)
  add_test(NAME color_api COMMAND glance_widgets_color)
  add_executable(glance_widgets_test tests/test_sdk.cpp)
  target_link_libraries(glance_widgets_test PRIVATE glance_widgets)
  add_test(NAME native_api COMMAND glance_widgets_test)
  file(GLOB GOLDEN_FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../spec/golden/*.json")
  if(GOLDEN_FILES)
    add_executable(glance_widgets_golden tests/test_golden.cpp)
    target_link_libraries(glance_widgets_golden PRIVATE glance_widgets)
    add_test(NAME golden COMMAND glance_widgets_golden ${GOLDEN_FILES})
  endif()
endif()
install(TARGETS glance_widgets EXPORT GlanceWidgetsTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT GlanceWidgetsTargets NAMESPACE GlanceWidgets:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GlanceWidgets)

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/GlanceWidgetsConfig.cmake"
  "include(\"\${CMAKE_CURRENT_LIST_DIR}/GlanceWidgetsTargets.cmake\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/GlanceWidgetsConfig.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GlanceWidgets)
